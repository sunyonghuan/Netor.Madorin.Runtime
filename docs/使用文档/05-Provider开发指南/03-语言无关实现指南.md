# 语言无关实现指南

> 使用任意编程语言实现 Provider Worker

本文档提供使用 Python、Go、Rust、Node.js 等语言实现 Provider Worker 的详细指南。

## 核心要求

要实现一个 Provider Worker，你的语言需要支持：

1. **读写二进制标准输入输出**（stdin/stdout）
2. **解析和生成 JSON**
3. **发起 HTTPS 请求**（调用 AI 服务 API）
4. **异步/并发处理**（可选但推荐）

## 帧协议实现

### 帧结构

每个帧由 4 字节大端长度前缀 + UTF-8 JSON 正文组成：

```
┌────────────────┬────────────────────────────┐
│  Length (4B)   │  JSON Body (Length bytes)  │
│  Big Endian    │  UTF-8                     │
└────────────────┴────────────────────────────┘
```

### Python 实现

```python
import sys
import json
import struct

MAX_FRAME_LENGTH = 16 * 1024 * 1024  # 16 MB

def write_frame(frame: dict):
    """写入一个帧到 stdout"""
    json_str = json.dumps(frame, ensure_ascii=False)
    body_bytes = json_str.encode('utf-8')
    
    if len(body_bytes) > MAX_FRAME_LENGTH:
        raise Exception(f"Frame too large: {len(body_bytes)}")
    
    # 大端 UInt32
    length_bytes = struct.pack('>I', len(body_bytes))
    
    sys.stdout.buffer.write(length_bytes)
    sys.stdout.buffer.write(body_bytes)
    sys.stdout.buffer.flush()

def read_frame() -> dict:
    """从 stdin 读取一个帧"""
    # 读取 4 字节长度
    length_bytes = sys.stdin.buffer.read(4)
    if len(length_bytes) == 0:
        return None  # EOF
    if len(length_bytes) < 4:
        raise Exception("Incomplete length prefix")
    
    length = struct.unpack('>I', length_bytes)[0]
    
    if length > MAX_FRAME_LENGTH:
        raise Exception(f"Frame too large: {length}")
    
    # 读取正文
    body_bytes = sys.stdin.buffer.read(length)
    if len(body_bytes) < length:
        raise Exception("Incomplete frame body")
    
    json_str = body_bytes.decode('utf-8')
    return json.loads(json_str)

# 使用示例
if __name__ == "__main__":
    # 读取 hello 帧
    hello = read_frame()
    
    # 发送 hello.ok 响应
    write_frame({
        "type": "hello.ok",
        "ipcVersion": 1,
        "slug": "myprovider",
        "protocol": "MyProvider"
    })
```

### Go 实现

```go
package main

import (
    "encoding/binary"
    "encoding/json"
    "fmt"
    "io"
    "os"
)

const MaxFrameLength = 16 * 1024 * 1024 // 16 MB

type Frame map[string]interface

func writeFrame(w io.Writer, frame Frame) error {
    jsonBytes, err := json.Marshal(frame)
    if err != nil {
        return err
    }
    
    if len(jsonBytes) > MaxFrameLength {
        return fmt.Errorf("frame too large: %d", len(jsonBytes))
    }
    
    // 大端 UInt32
    lengthBytes := make([]byte, 4)
    binary.BigEndian.PutUint32(lengthBytes, uint32(len(jsonBytes)))
    
    if _, err := w.Write(lengthBytes); err != nil {
        return err
    }
    
    if _, err := w.Write(jsonBytes); err != nil {
        return err
    }
    
    return nil
}

func readFrame(r io.Reader) (Frame, error) {
    // 读取 4 字节长度
    lengthBytes := make([]byte, 4)
    if _, err := io.ReadFull(r, lengthBytes); err != nil {
        if err == io.EOF {
            return nil, err
        }
        return nil, fmt.Errorf("failed to read length: %w", err)
    }
    
    length := binary.BigEndian.Uint32(lengthBytes)
    
    if length > MaxFrameLength {
        return nil, fmt.Errorf("frame too large: %d", length)
    }
    
    // 读取正文
    bodyBytes := make([]byte, length)
    if _, err := io.ReadFull(r, bodyBytes); err != nil {
        return nil, fmt.Errorf("failed to read body: %w", err)
    }
    
    var frame Frame
    if err := json.Unmarshal(bodyBytes, &frame); err != nil {
        return nil, err
    }
    
    return frame, nil
}

func main() {
    // 读取 hello 帧
    hello, err := readFrame(os.Stdin)
    if err != nil {
        fmt.Fprintf(os.Stderr, "Failed to read hello: %v\n", err)
        os.Exit(1)
    }
    
    // 发送 hello.ok 响应
    response := Frame{
        "type":       "hello.ok",
        "ipcVersion": 1,
        "slug":       "myprovider",
        "protocol":   "MyProvider",
    }
    
    if err := writeFrame(os.Stdout, response); err != nil {
        fmt.Fprintf(os.Stderr, "Failed to write hello.ok: %v\n", err)
        os.Exit(1)
    }
}
```

### Rust 实现

```rust
use std::io::{self, Read, Write};
use serde_json::Value;

const MAX_FRAME_LENGTH: u32 = 16 * 1024 * 1024; // 16 MB

fn write_frame<W: Write>(writer: &mut W, frame: &Value) -> io::Result<()> {
    let json_str = serde_json::to_string(frame)?;
    let body_bytes = json_str.as_bytes();
    
    if body_bytes.len() > MAX_FRAME_LENGTH as usize {
        return Err(io::Error::new(
            io::ErrorKind::InvalidInput,
            format!("Frame too large: {}", body_bytes.len()),
        ));
    }
    
    // 大端 UInt32
    let length = (body_bytes.len() as u32).to_be_bytes();
    
    writer.write_all(&length)?;
    writer.write_all(body_bytes)?;
    writer.flush()?;
    
    Ok(())
}

fn read_frame<R: Read>(reader: &mut R) -> io::Result<Option<Value>> {
    // 读取 4 字节长度
    let mut length_bytes = [0u8; 4];
    match reader.read_exact(&mut length_bytes) {
        Ok(_) => {},
        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
        Err(e) => return Err(e),
    }
    
    let length = u32::from_be_bytes(length_bytes);
    
    if length > MAX_FRAME_LENGTH {
        return Err(io::Error::new(
            io::ErrorKind::InvalidInput,
            format!("Frame too large: {}", length),
        ));
    }
    
    // 读取正文
    let mut body_bytes = vec![0u8; length as usize];
    reader.read_exact(&mut body_bytes)?;
    
    let json_str = String::from_utf8(body_bytes)
        .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
    
    let frame: Value = serde_json::from_str(&json_str)?;
    
    Ok(Some(frame))
}

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    let stdout = io::stdout();
    let mut stdin = stdin.lock();
    let mut stdout = stdout.lock();
    
    // 读取 hello 帧
    let hello = read_frame(&mut stdin)?
        .ok_or_else(|| io::Error::new(io::ErrorKind::UnexpectedEof, "No hello frame"))?;
    
    // 发送 hello.ok 响应
    let response = serde_json::json!({
        "type": "hello.ok",
        "ipcVersion": 1,
        "slug": "myprovider",
        "protocol": "MyProvider"
    });
    
    write_frame(&mut stdout, &response)?;
    
    Ok(())
}
```

### Node.js 实现

```javascript
const MAX_FRAME_LENGTH = 16 * 1024 * 1024; // 16 MB

function writeFrame(frame) {
    const jsonStr = JSON.stringify(frame);
    const bodyBuffer = Buffer.from(jsonStr, 'utf-8');
    
    if (bodyBuffer.length > MAX_FRAME_LENGTH) {
        throw new Error(`Frame too large: ${bodyBuffer.length}`);
    }
    
    // 大端 UInt32
    const lengthBuffer = Buffer.allocUnsafe(4);
    lengthBuffer.writeUInt32BE(bodyBuffer.length, 0);
    
    process.stdout.write(lengthBuffer);
    process.stdout.write(bodyBuffer);
}

async function readFrame() {
    // 读取 4 字节长度
    const lengthBuffer = await readExactly(4);
    if (lengthBuffer === null) {
        return null; // EOF
    }
    
    const length = lengthBuffer.readUInt32BE(0);
    
    if (length > MAX_FRAME_LENGTH) {
        throw new Error(`Frame too large: ${length}`);
    }
    
    // 读取正文
    const bodyBuffer = await readExactly(length);
    const jsonStr = bodyBuffer.toString('utf-8');
    
    return JSON.parse(jsonStr);
}

function readExactly(n) {
    return new Promise((resolve, reject) => {
        const chunks = [];
        let totalRead = 0;
        
        function onReadable() {
            let chunk;
            while ((chunk = process.stdin.read(n - totalRead)) !== null) {
                chunks.push(chunk);
                totalRead += chunk.length;
                
                if (totalRead >= n) {
                    process.stdin.removeListener('readable', onReadable);
                    process.stdin.removeListener('end', onEnd);
                    resolve(Buffer.concat(chunks));
                    return;
                }
            }
        }
        
        function onEnd() {
            process.stdin.removeListener('readable', onReadable);
            if (totalRead === 0) {
                resolve(null); // EOF
            } else {
                reject(new Error('Unexpected EOF'));
            }
        }
        
        process.stdin.on('readable', onReadable);
        process.stdin.on('end', onEnd);
        onReadable();
    });
}

// 使用示例
(async () => {
    // 读取 hello 帧
    const hello = await readFrame();
    
    // 发送 hello.ok 响应
    writeFrame({
        type: 'hello.ok',
        ipcVersion: 1,
        slug: 'myprovider',
        protocol: 'MyProvider'
    });
})();
```

## 完整 Worker 实现

### Python 完整示例

```python
#!/usr/bin/env python3
import sys
import json
import struct
import asyncio
import aiohttp
from typing import Dict, AsyncIterator

MAX_FRAME_LENGTH = 16 * 1024 * 1024

class ProviderWorker:
    def __init__(self, protocol: str, slug: str):
        self.protocol = protocol
        self.slug = slug
        self.active_invocations: Dict[str, asyncio.Task] = {}
    
    def write_frame(self, frame: dict):
        json_str = json.dumps(frame, ensure_ascii=False)
        body_bytes = json_str.encode('utf-8')
        
        if len(body_bytes) > MAX_FRAME_LENGTH:
            raise Exception(f"Frame too large: {len(body_bytes)}")
        
        length_bytes = struct.pack('>I', len(body_bytes))
        sys.stdout.buffer.write(length_bytes)
        sys.stdout.buffer.write(body_bytes)
        sys.stdout.buffer.flush()
    
    def read_frame(self) -> dict:
        length_bytes = sys.stdin.buffer.read(4)
        if len(length_bytes) == 0:
            return None
        if len(length_bytes) < 4:
            raise Exception("Incomplete length prefix")
        
        length = struct.unpack('>I', length_bytes)[0]
        
        if length > MAX_FRAME_LENGTH:
            self.write_frame({
                "type": "error",
                "invocationId": None,
                "code": "FrameTooLarge",
                "message": f"Frame length {length} exceeds {MAX_FRAME_LENGTH}"
            })
            sys.exit(2)
        
        body_bytes = sys.stdin.buffer.read(length)
        if len(body_bytes) < length:
            raise Exception("Incomplete frame body")
        
        json_str = body_bytes.decode('utf-8')
        return json.loads(json_str)
    
    def send_event(self, invocation_id: str, event: dict):
        self.write_frame({
            "type": "event",
            "invocationId": invocation_id,
            "event": event
        })
    
    def send_error(self, invocation_id: str, code: str, message: str):
        self.write_frame({
            "type": "error",
            "invocationId": invocation_id,
            "code": code,
            "message": message
        })
    
    async def handle_invoke(self, invoke: dict):
        invocation_id = invoke["invocationId"]
        connection = invoke["connection"]
        request = invoke["request"]
        
        try:
            async for event in self.call_ai_service(connection, request):
                self.send_event(invocation_id, event)
        except Exception as e:
            self.send_error(invocation_id, "InvokeFailed", str(e))
        finally:
            if invocation_id in self.active_invocations:
                del self.active_invocations[invocation_id]
    
    async def call_ai_service(self, connection: dict, request: dict) -> AsyncIterator[dict]:
        """调用 AI 服务并返回事件流"""
        api_url = connection["baseUrl"] + "/chat/completions"
        api_key = connection["apiKey"]
        
        headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        }
        
        body = {
            "model": request["model"],
            "messages": request["messages"],
            "stream": True
        }
        
        async with aiohttp.ClientSession() as session:
            async with session.post(api_url, json=body, headers=headers) as resp:
                if resp.status != 200:
                    error_text = await resp.text()
                    raise Exception(f"HTTP {resp.status}: {error_text}")
                
                # 解析 SSE 流
                async for line in resp.content:
                    line = line.decode('utf-8').strip()
                    if not line or not line.startswith('data: '):
                        continue
                    
                    data = line[6:]  # 去掉 "data: " 前缀
                    if data == '[DONE]':
                        break
                    
                    chunk = json.loads(data)
                    
                    # 提取文本增量
                    if 'choices' in chunk and chunk['choices']:
                        delta = chunk['choices'][0].get('delta', {})
                        if 'content' in delta:
                            yield {
                                "type": "output.delta",
                                "text": delta['content']
                            }
        
        # 发送完成事件
        yield {"type": "invocation.completed"}
    
    async def run(self):
        # 握手
        hello = self.read_frame()
        if hello is None or hello.get("type") != "hello":
            sys.exit(1)
        
        self.write_frame({
            "type": "hello.ok",
            "ipcVersion": 1,
            "slug": self.slug,
            "protocol": self.protocol
        })
        
        # 主循环
        while True:
            frame = self.read_frame()
            if frame is None:
                break
            
            frame_type = frame.get("type")
            
            if frame_type == "invoke":
                # 启动异步调用
                task = asyncio.create_task(self.handle_invoke(frame))
                self.active_invocations[frame["invocationId"]] = task
            
            elif frame_type == "cancel":
                invocation_id = frame["invocationId"]
                if invocation_id in self.active_invocations:
                    self.active_invocations[invocation_id].cancel()
            
            elif frame_type == "shutdown":
                # 等待所有调用完成
                if self.active_invocations:
                    await asyncio.gather(*self.active_invocations.values(), return_exceptions=True)
                break
            
            elif frame_type == "ping":
                self.write_frame({
                    "type": "pong",
                    "id": frame["id"]
                })
        
        return 0

def main():
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument('--slug', required=True)
    args = parser.parse_args()
    
    worker = ProviderWorker(protocol="MyProvider", slug=args.slug)
    
    try:
        exit_code = asyncio.run(worker.run())
        sys.exit(exit_code)
    except Exception as e:
        print(f"Fatal error: {e}", file=sys.stderr)
        sys.exit(2)

if __name__ == "__main__":
    main()
```

```go
import (
    "bufio"
    "net/http"
)

func callOpenAIAPI(baseURL, apiKey, model string, messages []Message) error {
    url := baseURL + "/chat/completions"
    
    body := map[string]interface{}{
        "model":    model,
        "messages": messages,
        "stream":   true,
    }
    
    req, _ := http.NewRequest("POST", url, jsonBody(body))
    req.Header.Set("Authorization", "Bearer "+apiKey)
    
    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        return err
    }
    defer resp.Body.Close()
    
    scanner := bufio.NewScanner(resp.Body)
    for scanner.Scan() {
        line := scanner.Text()
        // 处理 SSE 行
        handleSSELine(line)
    }
    
    return scanner.Err()
}
```

```rust
use reqwest::Client;
use futures::StreamExt;

async fn call_openai_api(
    base_url: &str,
    api_key: &str,
    model: &str,
    messages: Vec<Message>,
) -> Result<()> {
    let url = format!("{}/chat/completions", base_url);
    
    let body = serde_json::json!({
        "model": model,
        "messages": messages,
        "stream": true,
    });
    
    let client = Client::new();
    let resp = client
        .post(&url)
        .bearer_auth(api_key)
        .json(&body)
        .send()
        .await?;
    
    let mut stream = resp.bytes_stream();
    
    while let Some(chunk) = stream.next().await {
        let chunk = chunk?;
        // 处理 SSE 块
        handle_sse_chunk(&chunk);
    }
    
    Ok(())
}
```

## 安全实践

### API Key 脱敏

```python
def sanitize_api_key(key: str) -> str:
    """脱敏 API Key，只保留前缀和后缀"""
    if len(key) <= 8:
        return "***"
    return f"{key[:3]}...{key[-4:]}"

# 日志中使用
print(f"Using API Key: {sanitize_api_key(api_key)}", file=sys.stderr)
```

```python
try:
    response = await call_ai_service(connection, request)
except aiohttp.ClientError as e:
```
